def run_ablation(model, epochs=15, lr=7.3e-4, use_scheduler=True,
                 label_smoothing=0.08, weight_decay=3.1e-5,
                 patience=5, loader=None):
    """15-epoch ablation training on validation set."""
    train_ld = loader if loader else train_loader
    criterion = nn.CrossEntropyLoss(label_smoothing=label_smoothing)
    optimizer = torch.optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=lr, weight_decay=weight_decay)
    scheduler = (torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=epochs)
                 if use_scheduler else None)
    scaler = torch.cuda.amp.GradScaler() if DEVICE.type == 'cuda' else None
    es = EarlyStopping(patience=patience)

    for epoch in range(1, epochs + 1):
        train_one_epoch(model, train_ld, criterion, optimizer, scaler, DEVICE)
        _, val_f1_val, _, _, _ = evaluate(model, val_loader, criterion, DEVICE)
        if scheduler: scheduler.step()
        # use val_f1 as stopping signal via proxy
        vl_loss, vl_acc, _, _, _ = evaluate(model, val_loader, criterion, DEVICE)
        if es.step(vl_loss, model): break

    es.restore(model)
    _, val_acc, preds, labs, _ = evaluate(model, val_loader, criterion, DEVICE)
    val_f1 = f1_score(labs, preds, average='macro', zero_division=0)
    return val_acc, val_f1

# Expected Calibration Error (ECE)
correct   = (y_test_pred == y_test_true).astype(int)
max_probs = y_test_prob.max(axis=1)
n_bins    = 10
bin_edges = np.linspace(0, 1, n_bins + 1)
bin_accs, bin_confs, bin_cnts = [], [], []

for lo, hi in zip(bin_edges[:-1], bin_edges[1:]):
    mask = (max_probs >= lo) & (max_probs < hi)
    if mask.sum() > 0:
        bin_accs.append(correct[mask].mean())
        bin_confs.append(max_probs[mask].mean())
        bin_cnts.append(mask.sum())
    else:
        bin_accs.append(0); bin_confs.append((lo+hi)/2); bin_cnts.append(0)

ece = sum(c * abs(a - conf)
          for a, conf, c in zip(bin_accs, bin_confs, bin_cnts)) / len(y_test_true)

fig, axes = plt.subplots(1, 2, figsize=(14, 5))
fig.suptitle('Calibration Analysis', fontsize=13, fontweight='bold')
bin_labels = [f'{lo:.1f}' for lo in bin_edges[:-1]]

@torch.no_grad()
def eval_perturbed(model, loader, perturb_fn):
    model.eval()
    correct, total = 0, 0
    for imgs, labels in loader:
        perturbed = torch.stack([perturb_fn(img) for img in imgs]).to(DEVICE)
        preds = model(perturbed).argmax(dim=1)
        correct += (preds == labels.to(DEVICE)).sum().item()
        total   += len(labels)
    return correct / total

from src.utils import IMAGENET_MEAN, IMAGENET_STD

def jpeg_compress(img_tensor, q):
    mean_t = torch.tensor(IMAGENET_MEAN).view(3,1,1)
    std_t  = torch.tensor(IMAGENET_STD).view(3,1,1)
    pil = TF.to_pil_image((img_tensor * std_t + mean_t).clamp(0,1))
    buf = io.BytesIO()
    pil.save(buf, format='JPEG', quality=q)
    buf.seek(0)
    return TF.normalize(TF.to_tensor(Image.open(buf)), IMAGENET_MEAN, IMAGENET_STD)

for q in jpeg_quals:
    acc = eval_perturbed(final_model, test_loader,
        lambda img, q=q: jpeg_compress(img, q))

# Probe 4: FGSM
epsilons  = [0.0, 0.01, 0.02, 0.05, 0.10]
fgsm_accs = []
criterion_rob = nn.CrossEntropyLoss()

for eps in epsilons:
    correct, total = 0, 0
    for imgs, labels in test_loader:
        imgs   = imgs.to(DEVICE).requires_grad_(True)
        labels_dev = labels.to(DEVICE)
        if eps == 0:
            with torch.no_grad():
                preds = final_model(imgs).argmax(dim=1)
        else:
            loss = criterion_rob(final_model(imgs), labels_dev)
            loss.backward()
            perturbed = (imgs + eps * imgs.grad.sign()).detach()
            with torch.no_grad():
                preds = final_model(perturbed).argmax(dim=1)
        correct += (preds == labels_dev).sum().item()
        total   += len(labels)
        final_model.zero_grad()
    fgsm_accs.append(correct / total)
    print(f'  FGSM ε={eps:.2f}: {fgsm_accs[-1]:.4f}')
